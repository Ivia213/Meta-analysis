data <- read.csv("D:/R/data/MC2506/Data summary2506.csv", encoding = "UTF-8", fileEncoding = "UTF-8-BOM")

# 确保列名正确并转换数据类型

                     method = "FE"), silent = TRUE)

  stop("Fixed effects model did not fit properly.")

# 查看Egger检验的结果
print(egger_test)

# 执行Trim-and-Fill方法
taf_result <- trimfill(rma_model)

# 查看Trim-and-Fill分析的结果
print(taf_result)

# 绘制Trim-and-Fill调整后的森林图
forest(taf_result, main="Trim-and-Fill Adjusted Forest Plot")

# 计算失安全系数
# 使用已创建的rma模型对象
fsn_result <- fsn(rma_model, type = "Rosenthal")

# 安装并加载必要的包
if (!requireNamespace("readxl", quietly = TRUE)) install.packages("readxl")
if (!requireNamespace("metafor", quietly = TRUE)) install.packages("metafor")
library(readxl)
library(metafor)

# 读取Excel数据
data <- read.csv("D:/R/data/MC2506/Data summary2506.csv", encoding = "UTF-8", fileEncoding = "UTF-8-BOM")

# 检查列名
print(colnames(data))  # 确认列名为 Hedges..g.Effect.Size 和 Hedges..g.Standard.Error

# 转换为数值类型
data$Hedgesg_Effect_Size <- as.numeric(as.character(data$Hedges..g.Effect.Size))
data$Hedgesg_Standard_Error <- as.numeric(as.character(data$Hedges..g.Standard.Error))

# 检查数据结构
str(data)

# 创建固定效应模型
rma_model <- try(rma(yi = data$Hedgesg_Effect_Size, 
                     sei = data$Hedgesg_Standard_Error, 
                     data = data, 
                     method = "FE"), silent = TRUE)

if (inherits(rma_model, "try-error")) {
  stop("Fixed effects model did not fit properly.")
}

# Egger检验
egger_test <- regtest(rma_model, model = "rma")
print(egger_test)

# Trim and Fill
taf_result <- trimfill(rma_model)
print(taf_result)

# 设置标签
real_labels <- trimws(as.character(data$Author..year.))  # 去除前后空格
k_real <- length(real_labels)
k_total <- taf_result$k.f
k_filled <- k_total - k_real
filled_labels <- paste0("Filled ", seq_len(k_filled))
all_labels <- c(real_labels, filled_labels)

# 设置列标题
# 决定列位置（可以通过多次试验微调位置）
header_pos_x <- -5  # 第一列标题位置
effect_pos_x <- 2.8# 效应量列标题位置

# 绘图
forest(taf_result,
       slab = all_labels,
       main = "Trim-and-Fill Adjusted Forest Plot",
       xlab = "Hedges' g",
       refline = 0,
       cex = 0.9)

# 添加列标题
text(x = header_pos_x, y = taf_result$k.f + 2, labels = "Author (year)", font = 2)
text(x = effect_pos_x, y = taf_result$k.f + 2, labels = "Hedges' Effect Size and 95% CI", font = 2)
